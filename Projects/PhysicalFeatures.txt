import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

@SuppressWarnings("unused")
public class PhysicalFeatures {

  public int age;
  public String gender;
  public String name;
  public String sex;
  public String race;

  private String[] valid_races = {
    "black", "white", "Indian", "coloured", "hispanic"
  };

  private String[] valid_sex_genders = {
    "male", "female"
  };

  public PhysicalFeatures(String name, int age, String sex, String gender, String race) {
    // Validation conditions
    boolean case1 = name != null && !name.isEmpty();
    boolean case2 = age >= 0;
    boolean case3 = sex != null && Arrays.asList(valid_sex_genders).contains(sex.toLowerCase());
    boolean case4 = race != null && Arrays.asList(valid_races).contains(race.toLowerCase());

    // Map validation results to field names
    Map<Boolean, String> validationMap = new HashMap<>();
    validationMap.put(case1, "Name");
    validationMap.put(case2, "Age");
    validationMap.put(case3, "Sex");
    validationMap.put(case4, "Race");

    // Check for invalid cases
    for (Map.Entry<Boolean, String> entry : validationMap.entrySet()) {
      if (!entry.getKey()) {
        throw new IllegalArgumentException("Invalid " + entry.getValue());
      }
    }

    // Assign values if all validations pass
    this.name = name;
    this.age = age;
    this.sex = sex;
    this.gender = gender;
    this.race = race;
  }
}
